4. How to run it locally

# 1. Install Python deps
python -m pip install -r backend/requirements.txt

# 2. Run backend (with .env already on disk from the vault)
cd backend
uvicorn main:app --reload --port 8000

# 3. Run frontend
cd frontend
npm install --legacy-peer-deps   # three + react-three-fiber + drei already in package.json
npm run dev
# → visit http://localhost:3000/lounge  (3D world + voice Jarvis)
# → visit http://localhost:3000/observation (L1-L4 enforcement + envelopes)

cd <repo root>
python -m pytest backend/tests/ -q
